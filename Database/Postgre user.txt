sudo -i -u postgres
psql -U postgres -d scheduleLogic

CREATE ROLE schedulelogic_app LOGIN PASSWORD 'PASS'
    NOSUPERUSER NOCREATEDB NOCREATEROLE NOREPLICATION NOBYPASSRLS;

ALTER ROLE schedulelogic_app SET statement_timeout = '30s';
ALTER ROLE schedulelogic_app SET idle_in_transaction_session_timeout = '60s';

REVOKE CREATE ON SCHEMA public FROM PUBLIC;

GRANT CONNECT ON DATABASE "scheduleLogic" TO schedulelogic_app;
GRANT USAGE ON SCHEMA public TO schedulelogic_app;

GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO schedulelogic_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO schedulelogic_app;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
    GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO schedulelogic_app;
ALTER DEFAULT PRIVILEGES IN SCHEMA public
    GRANT USAGE, SELECT ON SEQUENCES TO schedulelogic_app;
